<script>
(function () {
  'use strict';

  var app = document.getElementById('talks-app');
  if (!app) return;

  function toggleTalk(id) {
    var btn = app.querySelector('[data-preview-talk="' + id + '"]');
    var viewer = document.getElementById('viewer-' + id);
    if (!btn || !viewer) return;

    var open = viewer.classList.toggle('is-open');
    btn.classList.toggle('is-active', open);
    btn.setAttribute('aria-expanded', open ? 'true' : 'false');
    btn.textContent = open ? 'Hide slides' : 'Preview slides';

    if (open) {
      if (!viewer.querySelector('iframe')) {
        var pdfUrl = btn.getAttribute('data-pdf');
        var iframe = document.createElement('iframe');
        iframe.src = pdfUrl;
        iframe.title = 'Talk slides PDF viewer';
        viewer.appendChild(iframe);
      }
    } else {
      viewer.innerHTML = '';
    }
  }

  app.querySelectorAll('[data-preview-talk]').forEach(function (btn) {
    btn.addEventListener('click', function () {
      toggleTalk(btn.getAttribute('data-preview-talk'));
    });
  });

  app.querySelectorAll('[data-thumb-for]').forEach(function (thumb) {
    thumb.addEventListener('click', function () {
      toggleTalk(thumb.getAttribute('data-thumb-for'));
    });
  });
})();
</script>
